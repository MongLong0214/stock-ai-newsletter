import {createReadStream} from 'node:fs'
import {readFile,writeFile,stat} from 'node:fs/promises'
import {createInterface} from 'node:readline'
import {createHash} from 'node:crypto'
import {execFileSync} from 'node:child_process'
import {rankLowVolatilityStableCandidates,passesCommonGate,LOW_VOLATILITY_STABLE_PARAMETERS} from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/strategies.ts'
import {hasCalculatedOutputMetrics} from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/generate-picks.ts'
const base='/tmp/composite-score-research-20260930/earlier-training'
const root='/Users/isaac/WebstormProjects/stock-ai-newsletter'
const sha=(x:string|Buffer)=>createHash('sha256').update(x).digest('hex')
async function main(){
 const manifest=JSON.parse(await readFile(`${base}/manifest.json`,'utf8'))
 const sourcePaths=['scripts/stock-picks/strategies.ts','scripts/stock-picks/generate-picks.ts','scripts/stock-picks/signals.ts']
 const sourceBefore=Object.fromEntries(await Promise.all(sourcePaths.map(async p=>[p,sha(await readFile(`${root}/${p}`))])))
 const mainSignalsHash=sha(execFileSync('git',['show','97b0c2ffa7b89a7ffdd4bae53e4c0e416e00f22f:scripts/stock-picks/signals.ts'],{cwd:root}))
 if(sourceBefore['scripts/stock-picks/signals.ts']!==manifest.sourceHashes['signals.ts']||sourceBefore['scripts/stock-picks/signals.ts']!==mainSignalsHash)throw new Error('Earlier signals are not the verified main implementation')
 const masters=new Map(manifest.currentMasters.map((m:any)=>[m.symbol,m]))
 const dates=manifest.output.dates as string[]
 const byDate=new Map(dates.map(date=>[date,[] as string[]]))
 const byDateCommon=new Map(dates.map(date=>[date,0]))
 const excluded=new Set<string>()
 const inputPath=manifest.output.path
 const inputBefore=await stat(inputPath)
 const inputHash=createHash('sha256')
 const stream=createReadStream(inputPath);stream.on('data',c=>inputHash.update(c))
 let rows=0,preCommon=0,common=0,eligible=0,storedEligible=0,preCommonFlagDifferences=0
 const differences:any[]=[]
 const visited=new Set<string>()
 for await(const line of createInterface({input:stream,crlfDelay:Infinity})){
  if(!line.trim())continue
  const row=JSON.parse(line);rows++
  if(!byDate.has(row.date)||!masters.has(row.symbol))throw new Error('Unrecognized earlier point')
  const key=`${row.date}|${row.symbol}`;if(visited.has(key))throw new Error(`Duplicate ${key}`);visited.add(key)
  const feature=row.feature
  const actualPreCommon=row.flags.currentKis&&hasCalculatedOutputMetrics(feature)
  if(actualPreCommon!==row.flags.preCommonPool)preCommonFlagDifferences++
  if(row.flags.priorTargetEligible)storedEligible++
  let actualEligible=false
  if(actualPreCommon){
   preCommon++
   if(passesCommonGate(feature,masters.get(row.symbol) as any,LOW_VOLATILITY_STABLE_PARAMETERS.minTurnover,LOW_VOLATILITY_STABLE_PARAMETERS.maxRsi)){
    common++;byDateCommon.set(row.date,byDateCommon.get(row.date)!+1)
   }
   // Membership is independent per feature. Singleton calls use the actual runtime gate
   // without allocating all 542k feature vectors or applying any label-based filter.
   actualEligible=rankLowVolatilityStableCandidates({features:[feature],masters:masters as any,parameters:LOW_VOLATILITY_STABLE_PARAMETERS,excludeSymbols:excluded,pickCount:1}).length===1
  }
  if(actualEligible){eligible++;byDate.get(row.date)!.push(row.symbol)}
  if(actualEligible!==row.flags.priorTargetEligible){
   differences.push({date:row.date,symbol:row.symbol,storedEligible:row.flags.priorTargetEligible,runtimeEligible:actualEligible,
    leftAssociatedDailyReturn:(1+feature.gapFromPreviousClosePercent/100)*feature.close/feature.open-1,
    runtimeParenthesizedDailyReturn:(1+feature.gapFromPreviousClosePercent/100)*(feature.close/feature.open)-1,
    intradayReturn:feature.close/feature.open-1})
  }
 }
 const rawInputHash=inputHash.digest('hex')
 if(rawInputHash!==manifest.output.sha256)throw new Error('Earlier input hash changed')
 const inputAfter=await stat(inputPath)
 if(inputBefore.size!==inputAfter.size||inputBefore.mtimeMs!==inputAfter.mtimeMs)throw new Error('Earlier input mutated during export')
 const outputText=dates.map(date=>JSON.stringify({date,runtimeEligibleSymbols:byDate.get(date)!.sort()})).join('\n')+'\n'
 const outputPath=`${base}/earlier-runtime-eligibility.ndjson`
 await writeFile(outputPath,outputText,{flag:'wx'})
 const sourceAfter=Object.fromEntries(await Promise.all(sourcePaths.map(async p=>[p,sha(await readFile(`${root}/${p}`))])))
 const audit={createdAt:new Date().toISOString(),sourceBefore,sourceAfter,sourceChanged:sourcePaths.filter(p=>sourceBefore[p]!==sourceAfter[p]),
  actualFunction:'rankLowVolatilityStableCandidates',membershipCall:'singleton feature, pickCount=1, excludeSymbols empty; actual independent passesTargetPoolGate contract',
  parameters:LOW_VOLATILITY_STABLE_PARAMETERS,basePool:'currentKis && actual hasCalculatedOutputMetrics; no future label availability or outcome filter',
  input:{path:inputPath,sha256:rawInputHash,rows,unchanged:true},output:{path:outputPath,sha256:sha(outputText),dateRows:dates.length,eligibleMemberships:eligible},
  counts:{preCommon,commonGate:common,runtimeEligible:eligible,storedEligible,preCommonFlagDifferences,storedEligibilityDifferences:differences.length},differences,
  signalProvenance:{version:manifest.signalScoreVersion,earlierHash:manifest.sourceHashes['signals.ts'],main97b0c2fHash:mainSignalsHash,currentHash:sourceBefore['scripts/stock-picks/signals.ts'],allEqual:true},
  perDate:dates.map(date=>({date,preReturnCommonGate:byDateCommon.get(date),runtimeEligibleCount:byDate.get(date)!.length})),
  earlierOutputsModified:false,databaseCalls:0,futureLabelFieldsUsed:false,
 }
 await writeFile(`${base}/earlier-runtime-eligibility-audit.json`,JSON.stringify(audit,null,2)+'\n',{flag:'wx'})
 console.log(JSON.stringify({output:audit.output,counts:audit.counts,differences,sourceChanged:audit.sourceChanged,signalProvenance:audit.signalProvenance}))
}
main().catch(e=>{console.error(e);process.exitCode=1})
