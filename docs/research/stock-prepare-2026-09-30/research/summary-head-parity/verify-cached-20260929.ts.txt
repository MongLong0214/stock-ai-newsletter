import assert from 'node:assert/strict'
import { readFileSync, writeFileSync } from 'node:fs'
import { createHash } from 'node:crypto'
import { buildSignals } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/signals.ts'
import { rankLowVolatilityStableCandidates, LOW_VOLATILITY_STABLE_PARAMETERS } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/strategies.ts'
import { PRODUCTION_STRATEGY } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/production-strategy.ts'
import { buildSignals as headSignals } from './signals-head97b.ts'
import { rankLowVolatilityStableCandidates as headRank } from './strategies-head97b.ts'
const sourcePath='/tmp/stock-upside-today-candidates.json'
const text=readFileSync(sourcePath,'utf8')
const cached=JSON.parse(text)
assert.equal(cached.signalDate,'2026-09-29')
const metadata=JSON.parse(readFileSync('/tmp/stock-target-research-20260929/metadata.json','utf8'))
const features=cached.points.map((p:any)=>p.feature)
const args={features,masters:new Map(metadata.masters.map((master:any)=>[master.symbol,master])),parameters:LOW_VOLATILITY_STABLE_PARAMETERS,excludeSymbols:new Set<string>(),pickCount:features.length}
const head=headRank(args)
const current=rankLowVolatilityStableCandidates(args)
assert.deepEqual(current,head)
for(const feature of features)assert.deepEqual(buildSignals(feature),headSignals(feature))
const selected=current.slice(0,3).map(symbol=>{
 const f=features.find((feature:any)=>feature.symbol===symbol)
 return {symbol,atrScore:f.atrPercent14,signals:buildSignals(f)}
})
const report={signalDate:cached.signalDate,scope:'actual 9/29 cached candidate features; full 320-bar input and original cooldown exclusions were not recaptured',sourcePath,sourceSha256:createHash('sha256').update(text).digest('hex'),featureRows:features.length,rankedRows:current.length,sameFullSelectorOrder:true,sameAllSevenNumericSignals:true,selected,productionStrategy:PRODUCTION_STRATEGY.name,productionVersion:PRODUCTION_STRATEGY.version,parametersHash:PRODUCTION_STRATEGY.parametersHash}
writeFileSync('/tmp/composite-score-research-20260930/summary-head-parity/cached-20260929-feature-evidence.json',JSON.stringify(report,null,2))
console.log(JSON.stringify(report))
