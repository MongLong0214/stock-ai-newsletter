import assert from 'node:assert/strict'
import { readFileSync, writeFileSync } from 'node:fs'
import { createHash } from 'node:crypto'
import { addKoreanTradingDays } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/lib/tli/trading-calendar.ts'
import { buildPriceBook } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/data-handler.ts'
import { TradingDayIndex } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/trading-days.ts'
import { generatePicksWithMeta, type StockPickMaster } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/generate-picks.ts'
import { generatePicksWithMeta as generateHead } from './generate-picks-head97b.ts'
import type { StockDailyPriceRow } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/tli/prices/stock-daily-prices.ts'
const SIGNAL_DATE = '2026-08-27'
const TODAY_KST = '2026-08-28'
const SYMBOLS = ['KOSPI:000010', 'KOSPI:000020', 'KOSDAQ:000030'] as const
const FOUR_SYMBOLS = [...SYMBOLS, 'KOSPI:000040'] as const

const makeFixture = (symbols: readonly string[] = SYMBOLS, signalDate = SIGNAL_DATE) => {
  const dates = Array.from(
    { length: 320 },
    (_value, index) => addKoreanTradingDays(signalDate, index - 319),
  )
  const rows: StockDailyPriceRow[] = symbols.flatMap((symbol, symbolIndex) => dates.map((tradeDate, index) => {
    const base = 2_000 + symbolIndex * 100
    const isSignalDay = index === dates.length - 1
    // 프로덕션 게이트(excludeGapUp·60일 신고가 돌파)를 통과하는 신호일 캔들:
    // 시가 = 전일 종가(갭 0), 종가 = 직전 최고 종가(+10 홀수일 보정 포함) 대비 +0.3% 돌파
    const previousClose = base + (index - 1) * 2 + ((index - 1) % 2 === 1 ? 10 : -10)
    const priorMaxClose = base + (dates.length - 3) * 2 + 10
    const close = isSignalDay
      ? Math.round(priorMaxClose * (symbolIndex === 2 ? 0.995 : 1.003))
      : base + index * 2 + (index % 2 === 1 ? 10 : -10)
    return {
      symbol,
      trade_date: tradeDate,
      open: isSignalDay ? previousClose : close - 5,
      high: Math.max(close, isSignalDay ? previousClose : close - 5) + 15,
      low: Math.min(close, isSignalDay ? previousClose : close - 5) - 15,
      close,
      volume: isSignalDay ? 5_000_000 + symbolIndex * 100_000 : 1_000_000 + index * 1_000,
      source: 'kis',
    }
  }))
  const masters: StockPickMaster[] = symbols.map((symbol, index) => ({
    symbol,
    name: `테스트종목${index + 1}`,
    is_active: true,
    status_flags: {},
  }))
  return { dates, rows, prices: buildPriceBook(rows), masters }
}


async function main() {
  const fixture=makeFixture(FOUR_SYMBOLS)
  const rows=fixture.rows.map(row=> {
    if(row.trade_date!==SIGNAL_DATE)return row
    if(row.symbol===FOUR_SYMBOLS[0])return {...row,volume:10000,
      high:Math.max(row.close!,row.open!)+5,low:Math.min(row.close!,row.open!)-5}
    if(row.symbol===FOUR_SYMBOLS[3])return {...row,
      high:Math.max(row.close!,row.open!)+400,low:Math.min(row.close!,row.open!)-400}
    return row
  })
  const benchmarkRows=fixture.dates.map(trade_date=>({symbol:'KOSPI',trade_date,
    open:1000,high:1005,low:995,close:1000,volume:1000000,source:'kis' as const}))
  const dependencies={
    loadTradingDays:async()=>new TradingDayIndex(fixture.dates),
    loadPrices:async()=>buildPriceBook([...rows,...benchmarkRows]),
    loadMasters:async()=>fixture.masters,
    loadRecentPublishedSymbols:async()=>new Set<string>(),
  }
  const [baseline,current]=await Promise.all([
    generateHead({todayKst:TODAY_KST,dependencies}),
    generatePicksWithMeta({todayKst:TODAY_KST,dependencies}),
  ])
  const withoutSummary=(picks: typeof current.picks)=>picks.map(({rationale,...pick})=>pick)
  assert.deepEqual(withoutSummary(current.picks),withoutSummary(baseline.picks))
  assert.deepEqual(current.meta,baseline.meta)
  assert(current.picks.every(pick=>pick.rationale.split('|').length===18))
  assert(current.picks.every(pick=>!/변동성 안정 순위|선정 경로 저변동 안정/.test(pick.rationale)))
  assert(baseline.picks.every(pick=>/변동성 안정 순위|선정 경로 저변동 안정/.test(pick.rationale)))
  const evidence={head:'97b0c2ffa7b89a7ffdd4bae53e4c0e416e00f22f',
    rawFixture:{sessions:320,symbols:4,rows:rows.length,sha256:createHash('sha256').update(JSON.stringify(rows)).digest('hex')},
    sameSelectionAndAllNumericSignals:true,sameCompleteMetadataAndShadowSelections:true,
    selections:current.picks.map(pick=>({symbol:pick.ticker,signals:pick.signals,selection:pick.selection})),
    ranking:current.meta.rankedCandidates.map(pick=>({symbol:pick.symbol,atrScore:pick.score,tier:pick.tier})),
    strategy:current.meta.strategy,strategyVersion:current.meta.strategyVersion,parametersHash:current.meta.parametersHash,
    summaryItems:current.picks.map(pick=>pick.rationale.split('|').length),newSummaries:current.picks.map(pick=>pick.rationale)}
  writeFileSync('/tmp/composite-score-research-20260930/summary-head-parity/evidence.json',JSON.stringify(evidence,null,2))
  console.log(JSON.stringify({event:'summary_head97b_parity',...evidence,newSummaries:undefined}))
}
main().catch(error=>{console.error(error);process.exitCode=1})
