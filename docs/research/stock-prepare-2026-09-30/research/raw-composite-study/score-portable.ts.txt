// Research artifact only. Parameters are the frozen TRAIN fit; no runtime fitting.
export interface RawNormalizer {
  uniqueValues: readonly number[]
  midpointProbabilities: readonly number[]
}
export interface RawUtilityModel {
  intercept: number
  coefficients: readonly number[]
  normalizers: readonly RawNormalizer[]
}
export interface RawScoreFeature {
  open: number | null
  close: number | null
  gapFromPreviousClosePercent: number | null
  volumeRatio20: number | null
  atrPercent14: number | null
}
export interface RawScoreContext {
  chaikinMoneyFlow21: number | null
  distanceFromPriorHigh20Percent: number | null
  bollingerWidth20Percent: number | null
}
const valid = (v: unknown): v is number => typeof v === 'number' && Number.isFinite(v)
export function observedAtoms(feature: RawScoreFeature, context: RawScoreContext): number[] | null {
  if (!valid(feature.open) || feature.open <= 0 || !valid(feature.close)
    || !valid(feature.gapFromPreviousClosePercent)) return null
  const change = ((1 + feature.gapFromPreviousClosePercent / 100) * (feature.close / feature.open) - 1) * 100
  const values = [context.chaikinMoneyFlow21, context.distanceFromPriorHigh20Percent,
    feature.volumeRatio20, feature.atrPercent14, context.bollingerWidth20Percent, Math.max(0, change)]
  return values.every(valid) ? values as number[] : null
}
export function normalizeObserved(value: number, curve: RawNormalizer): number {
  const x = curve.uniqueValues
  const y = curve.midpointProbabilities
  if (value < x[0]!) return 0
  if (value > x[x.length - 1]!) return 1
  for (let i = 0; i < x.length; i++) {
    if (value === x[i]) return y[i]!
    if (i > 0 && value < x[i]!) {
      return y[i - 1]! + (value - x[i - 1]!) / (x[i]! - x[i - 1]!) * (y[i]! - y[i - 1]!)
    }
  }
  throw new Error('Invalid research normalizer curve')
}
export function scoreRawUtility(atoms: readonly number[], model: RawUtilityModel): {score: number; mu: number} | null {
  if (atoms.length !== 6 || !atoms.every(valid)) return null
  const signs = [1, 1, 1, 1, -1, -1]
  let mu = model.intercept
  for (let j = 0; j < 6; j++) {
    mu += signs[j]! * model.coefficients[j]! * normalizeObserved(atoms[j]!, model.normalizers[j]!)
  }
  const score = Math.floor(Math.min(100, Math.max(0, 50 + 500 * mu)) + 0.5)
  return {score, mu}
}
