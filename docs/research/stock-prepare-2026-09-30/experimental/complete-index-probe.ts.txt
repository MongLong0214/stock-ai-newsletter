import {createRequire} from 'node:module'
import {readFile,writeFile} from 'node:fs/promises'
import {createHash} from 'node:crypto'
const root='/Users/isaac/WebstormProjects/stock-ai-newsletter',out='/tmp/composite-score-experimental-20260930',require=createRequire(root+'/package.json')
require('dotenv').config({path:root+'/.env.local',quiet:true})
const fetchOriginal=globalThis.fetch,calls:any[]=[];let blockedWrites=0
// Only token acquisition may POST; price/calendar evidence and storage are otherwise read-only.
globalThis.fetch=async(input,init)=>{const url=new URL(input instanceof Request?input.url:String(input));const method=(init?.method??(input instanceof Request?input.method:'GET')).toUpperCase();if(!['GET','HEAD'].includes(method)&&!(url.hostname==='openapi.koreainvestment.com'&&url.pathname==='/oauth2/tokenP'&&method==='POST')){blockedWrites++;throw new Error('Read-only exploration blocked external mutation')}
const started=Date.now(),r=await fetchOriginal(input,init);if(url.pathname.includes('chartprice')){const b:any=await r.clone().json();calls.push({symbolCode:url.searchParams.get('FID_INPUT_ISCD'),start:url.searchParams.get('FID_INPUT_DATE_1'),end:url.searchParams.get('FID_INPUT_DATE_2'),status:r.status,elapsedMs:Date.now()-started,rows:b.output2?.length,rt_cd:b.rt_cd,msg_cd:b.msg_cd})}return r}
const previous=(d:string)=>{const x=new Date(d+'T00:00:00Z');x.setUTCDate(x.getUTCDate()-1);return x.toISOString().slice(0,10).replaceAll('-','')}
async function main(){const start=Date.now();const report=JSON.parse(await readFile(out+'/long-history-probe.json','utf8'));const data=(await readFile(out+'/long-history-sample.ndjson','utf8')).trim().split('\n').map(s=>JSON.parse(s));const index=data.find(([s])=>s==='KOSPI');const byDate=new Map(index[1].map((r:any)=>[r.trade_date,r]));const {fetchIndexDailyRangePriceRows}=await import(root+'/app/archive/_utils/api/kis/client.ts');let end=previous(index[1][0].trade_date),pages=0
while(end>='20220101'){const st=Date.now();const rows=await fetchIndexDailyRangePriceRows('0001','20220101',end);pages++;const valid=rows.filter(r=>r.date.replaceAll('-','')>='20220101'&&r.date.replaceAll('-','')<=end).sort((a,b)=>a.date.localeCompare(b.date));if(!valid.length)break;for(const r of valid)byDate.set(r.date,{symbol:'KOSPI',trade_date:r.date,open:r.open,high:r.high,low:r.low,close:r.close,volume:r.volume,source:'kis'});const next=previous(valid[0]!.date);if(next>=end)throw new Error('No backward progress');end=next;if(pages>20)throw new Error('Index page cap');await new Promise(r=>setTimeout(r,Math.max(0,550-(Date.now()-st))))}
index[1]=[...byDate.values()].sort((a:any,b:any)=>a.trade_date.localeCompare(b.trade_date));const sample=report.samples.find((s:any)=>s.symbol==='KOSPI');sample.rows=index[1].length;sample.pages+=pages;sample.firstDate=index[1][0].trade_date;sample.lastDate=index[1].at(-1).trade_date
const samsungDates=new Set(data.find(([s])=>s==='KOSPI:005930')[1].map((r:any)=>r.trade_date));const indexDates=new Set(index[1].map((r:any)=>r.trade_date));const gaps={indexMissingVsSamsung:[...samsungDates].filter(d=>!indexDates.has(d)),indexExtraVsSamsung:[...indexDates].filter(d=>!samsungDates.has(d))}
const text=data.map(r=>JSON.stringify(r)).join('\n')+'\n';report.calls.push(...calls);report.totalPriceCalls=report.calls.length;report.totalRows=data.reduce((n,[,r])=>n+r.length,0);report.blockedWrites+=blockedWrites;report.indexContinuation={pages,elapsedSeconds:(Date.now()-start)/1000,terminatedBy:'no rows within remaining bounded range',observedPageSizes:[...new Set(calls.map(c=>c.rows))],calendarParity:gaps};report.rawSha256=createHash('sha256').update(text).digest('hex');report.finishedAt=new Date().toISOString();report.finalRawPath=out+'/long-history-sample-complete.ndjson'
await writeFile(report.finalRawPath,text);await writeFile(out+'/long-history-probe-complete.json',JSON.stringify(report,null,2)+'\n');console.log(JSON.stringify({samples:report.samples,totalPriceCalls:report.totalPriceCalls,totalRows:report.totalRows,indexContinuation:report.indexContinuation,adjustmentProbe:report.adjustedVersusUnadjustedProbe,blockedWrites:report.blockedWrites,rawFieldNames:report.rawFieldNames}))}
main().catch(e=>{console.error(e.message);process.exitCode=1})
