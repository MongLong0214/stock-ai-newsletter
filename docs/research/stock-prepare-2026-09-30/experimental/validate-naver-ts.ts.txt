import { readFileSync, writeFileSync } from 'node:fs'
import { createHash } from 'node:crypto'
import { deepStrictEqual, strictEqual, ok, throws } from 'node:assert'
import { buildFeatureVector } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/features.ts'
import { buildTechnicalContextMap } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/technical-context.ts'
import { buildSignals } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/signals.ts'
import { rankLowVolatilityStableCandidates, LOW_VOLATILITY_STABLE_PARAMETERS, type StockMasterState } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/strategies.ts'
import { hasCalculatedOutputMetrics } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/generate-picks.ts'
import { LookaheadError } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/data-handler.ts'
import { TradingDayIndex } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/trading-days.ts'
import { vendorHandler, computeObservedSymbol, originalInputs18, currentProductionEligible, relaxedMomentumGate, type VendorRow } from './naver-ts-observed.ts'

const base = '/tmp/composite-score-experimental-20260930', root = '/Users/isaac/WebstormProjects/stock-ai-newsletter'
const sha = (x: string | Buffer) => createHash('sha256').update(x).digest('hex')
globalThis.fetch = async () => { throw new Error('Offline TS validation blocks network') }
const sourceFiles = ['features.ts', 'technical-context.ts', 'indicators.ts', 'data-handler.ts', 'data-contract.ts', 'signals.ts', 'strategies.ts', 'generate-picks.ts']
const hashes = () => Object.fromEntries(sourceFiles.map(f => [f, sha(readFileSync(`${root}/scripts/stock-picks/${f}`))]))
const before = hashes()
const path = `${base}/naver-history-sample-four.ndjson`, bytes = readFileSync(path)
const groups: Array<[string, VendorRow[]]> = bytes.toString().trim().split('\n').map(line => JSON.parse(line))
const benchmark = groups.find(([symbol]) => symbol === 'KOSPI')![1]
const symbols = groups.filter(([symbol]) => symbol !== 'KOSPI').map(([symbol]) => symbol)
const masterInput = JSON.parse(readFileSync(`${base}/naver-history/master-snapshot.json`, 'utf8'))
const masters = new Map<string, StockMasterState>(masterInput.masters.map((m: StockMasterState) => [m.symbol, m]))
const index = new TradingDayIndex(benchmark.map(row => row.trade_date))
const firstSignal = index.firstTradingDayOnOrAfter('2020-01-01')!
const lastSignal = index.tradingDays.filter(d => d <= '2024-12-31').at(-1)!
const historyDates = index.tradingDays.slice(index.indexByDate.get(firstSignal)! - 319, index.indexByDate.get(lastSignal)! + 1)
ok(historyDates.length > 320)
const complete = new Map(symbols.map(symbol => {
  const rows = groups.find(([s]) => s === symbol)![1]
  return [symbol, computeObservedSymbol({ master: masters.get(symbol)!, rows, benchmark, index, historyDates, includeFromDate: firstSignal })]
}))
const witnesses: unknown[] = [], output: unknown[] = []
let sourceParity = 0, futureParity = 0, directFutureBlocked = 0, gateParity = 0, breadthParity = 0
const dates = [firstSignal, '2020-06-30', '2021-06-30', '2022-12-29', '2023-06-30', '2024-12-30']
for (const date of dates) {
  const prefixDates = index.tradingDays.slice(Math.max(0, index.indexByDate.get(date)! - 319), index.indexByDate.get(date)! + 1)
  const prefixGroups = groups.map(([symbol, rows]) => [symbol, rows.filter(r => r.trade_date <= date)] as const)
  const mutatedGroups = groups.map(([symbol, rows]) => [symbol, rows.map(row => row.trade_date > date ? {
    ...row, open: row.open! * 100, high: row.high! * 100, low: row.low! * 100, close: row.close * 100, volume: (row.volume ?? 0) * 100,
  } : row)] as const)
  const hp = vendorHandler(prefixGroups, index, date), hf = vendorHandler(mutatedGroups, index, date)
  const cp = buildTechnicalContextMap({ handler: hp, symbols, dates: prefixDates, includeFromDate: date }).get(date)!
  const cf = buildTechnicalContextMap({ handler: hf, symbols, dates: prefixDates, includeFromDate: date }).get(date)!
  throws(() => hf.get(symbols[0]!, index.nextTradingDay(date, 1)!), LookaheadError); directFutureBlocked++
  const actualFeatures = symbols.map(symbol => buildFeatureVector(hp, symbol, date))
  const currentPool = actualFeatures.filter(hasCalculatedOutputMetrics)
  const eligible = new Set(rankLowVolatilityStableCandidates({ features: currentPool, masters, parameters: LOW_VOLATILITY_STABLE_PARAMETERS, excludeSymbols: new Set(), pickCount: symbols.length }))
  const oneEligible = symbols.map(symbol => complete.get(symbol)!.find(p => p.feature.simDate === date)!)
  const eligibleBreadth = oneEligible.reduce((sum, p) => sum + p.context.breadthEligibleSymbols, 0)
  const aboveBreadth = oneEligible.reduce((sum, p) => sum + (p.context.breadthAboveSma20 === 1 ? 1 : 0), 0)
  for (const [i, symbol] of symbols.entries()) {
    const feature = actualFeatures[i]!, futureFeature = buildFeatureVector(hf, symbol, date)
    const original = originalInputs18(feature, cp.get(symbol)!), future = originalInputs18(futureFeature, cf.get(symbol)!)
    deepStrictEqual(original, future); deepStrictEqual(buildSignals(feature), buildSignals(futureFeature)); futureParity++
    const seriesPoint = complete.get(symbol)!.find(p => p.feature.simDate === date)!
    deepStrictEqual(feature, seriesPoint.feature)
    const mergedOriginal = [...seriesPoint.originalInputs18]
    mergedOriginal[17] = eligibleBreadth ? aboveBreadth / eligibleBreadth * 100 : null
    deepStrictEqual(original, mergedOriginal); deepStrictEqual(buildSignals(feature), seriesPoint.sourceSignals); sourceParity++
    strictEqual(seriesPoint.runtimeEligible, eligible.has(symbol)); gateParity++
    strictEqual(cp.get(symbol)!.breadthEligibleSymbols, eligibleBreadth)
    strictEqual(cp.get(symbol)!.breadthUniverseSymbols, symbols.length)
    strictEqual(cp.get(symbol)!.breadthAboveSma20, eligibleBreadth ? aboveBreadth / eligibleBreadth : null); breadthParity++
    const raw = groups.find(([s]) => s === symbol)![1].find(r => r.trade_date === date)!
    strictEqual(currentProductionEligible(feature, { ...masters.get(symbol)!, status_flags: { managed_stock: 'Y' } }, raw), false)
    strictEqual(currentProductionEligible(feature, { ...masters.get(symbol)!, is_active: false }, raw), false)
    strictEqual(currentProductionEligible(feature, masters.get(symbol)!, { ...raw, source: 'kis' } as unknown as VendorRow), false)
    output.push({ date, symbol, feature, context: cp.get(symbol), originalInputs18: original, sourceSignals: buildSignals(feature), runtimeEligible: eligible.has(symbol), relaxedMomentumGate: seriesPoint.relaxedMomentumGate })
    witnesses.push({ date, symbol, rawSource: raw.source, source18AndScoresEqual: true, futureSuffixEqual: true, eligibilityEqualsActualSelector: true, marketBreadthEqualsActualThreeSymbolMap: true })
  }
}
// Meaningful guard witnesses retain every valid actual source metric except the explicit observed momentum fields.
const validPoint = [...complete.values()].flat().find(p => p.runtimeEligible)!
ok(validPoint)
const master = masters.get(validPoint.feature.symbol)!, raw = groups.find(([s]) => s === master.symbol)![1].find(r => r.trade_date === validPoint.feature.simDate)!
const rsi80 = { ...validPoint.feature, rsi14: 80 }
strictEqual(currentProductionEligible(rsi80, master, raw), false); strictEqual(relaxedMomentumGate(rsi80, master, raw), true)
const day14 = { ...validPoint.feature, open: validPoint.feature.close, gapFromPreviousClosePercent: 14 }
strictEqual(currentProductionEligible(day14, master, raw), false); strictEqual(relaxedMomentumGate(day14, master, raw), true)
strictEqual(relaxedMomentumGate({ ...rsi80, rsi14: 86 }, master, raw), false)
strictEqual(relaxedMomentumGate({ ...day14, gapFromPreviousClosePercent: 16 }, master, raw), false)
throws(() => vendorHandler(groups.map(([s, rows]) => [s, [{ ...rows[0]!, source: 'kis' }, ...rows.slice(1)]] as unknown as [string, VendorRow[]]), index, lastSignal), /Wrong provider/)
const after = hashes(); deepStrictEqual(before, after)
const serialized = output.map(p => JSON.stringify(p)).join('\n') + '\n'
writeFileSync(`${base}/naver-ts-sample-observed.ndjson`, serialized)
const audit = {
  passed: true, sampleInputPath: path, sampleInputSha256: sha(bytes), moduleSha256: sha(readFileSync(`${base}/naver-ts-observed.ts`)), sourceHashes: before, sourceChanged: false,
  sampledSymbols: symbols, calendarRows: index.tradingDays.length, full320Warm: index.indexByDate.get(firstSignal)! >= 319,
  dates, witnesses, fullFeatureVectorIncludingAtrPercentile60Parity: sourceParity, source18AndSevenScoreParity: sourceParity, futureSuffixParity: futureParity, directFutureReadsBlocked: directFutureBlocked,
  actualCurrentSelectorMembershipParity: gateParity, exactMarketBreadthCompositionParity: breadthParity,
  providerTagsPreserved: true, wrongProviderRejected: true, currentSourceAcceptanceExplicit: 'naver-fchart offline only',
  originalCurrentGateRelaxedGateNotMixed: true, relaxedObservationGuardWitnesses: 4,
  rawPricesChanged: false, labelReads: 0, fitCalls: 0, stockSelections: 0, networkCalls: 0,
  sampleBreadthUniverseOnly: 3, productionBulkBreadthMustUseAllActiveMasterSymbols: true,
  output: { path: `${base}/naver-ts-sample-observed.ndjson`, sha256: sha(serialized), rows: output.length },
}
writeFileSync(`${base}/naver-ts-sample-audit.json`, JSON.stringify(audit, null, 2) + '\n')
console.log(JSON.stringify(audit))
