import { createReadStream, readFileSync, writeFileSync, openSync, readSync, closeSync } from 'node:fs'
import { createInterface } from 'node:readline'
import { createHash } from 'node:crypto'
import { deepStrictEqual, strictEqual } from 'node:assert'
import { buildFeatureVector } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/features.ts'
import { buildTechnicalContextMap } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/technical-context.ts'
import { buildSignals } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/signals.ts'
import { TradingDayIndex } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/trading-days.ts'
import { vendorHandler, originalInputs18, currentProductionEligible, SIGNAL_NAMES, type VendorRow } from './naver-ts-observed.ts'

const base = '/tmp/composite-score-experimental-20260930', root = '/Users/isaac/WebstormProjects/stock-ai-newsletter'
globalThis.fetch = async () => { throw new Error('Offline NAVER audit blocks network') }
const sha = (x: Buffer) => createHash('sha256').update(x).digest('hex')
async function main() {
  const manifestPath = `${base}/naver-ts-observed/manifest.json`, m = JSON.parse(readFileSync(manifestPath, 'utf8'))
  if (m.sampleOnly || m.source !== 'naver-fchart' || m.output.columns !== 28) throw new Error('Audit requires completed full observed artifact')
  const masters = new Map(JSON.parse(readFileSync(m.master.path, 'utf8')).masters.map((x: { symbol: string }) => [x.symbol, x]))
  const index = new TradingDayIndex(m.calendarDates)
  const fd = openSync(m.output.path, 'r'), buf = Buffer.alloc(28 * 8)
  const samples: Array<{ row: number; symbol: string; date: string; values: number[] }> = []
  for (let j = 0; j < 100; j++) {
    const row = Math.floor(j * (m.output.rows - 1) / 99)
    if (readSync(fd, buf, 0, buf.length, row * buf.length) !== buf.length) throw new Error('Truncated observed artifact')
    const values = Array.from({ length: 28 }, (_, i) => buf.readDoubleLE(i * 8))
    samples.push({ row, date: m.signalDates[values[0]!], symbol: m.symbols[values[1]!], values })
  }
  closeSync(fd)
  const wanted = new Set(samples.map(p => p.symbol)), groups = new Map<string, VendorRow[]>()
  for await (const line of createInterface({ input: createReadStream(m.raw.path), crlfDelay: Infinity })) {
    const [symbol, rows] = JSON.parse(line)
    if (symbol === 'KOSPI' || wanted.has(symbol)) groups.set(symbol, rows)
  }
  const benchmark = groups.get('KOSPI')!, witnesses: unknown[] = []
  for (const p of samples) {
    const rows = groups.get(p.symbol)!, dateIndex = index.indexByDate.get(p.date)!
    const dates = index.tradingDays.slice(dateIndex - 319, dateIndex + 1)
    const mutation = (r: VendorRow): VendorRow => ({ ...r, open: r.open! * 100, high: r.high! * 100, low: r.low! * 100, close: r.close * 100, volume: (r.volume ?? 0) * 100 })
    const point = (stockRows: VendorRow[], benchmarkRows: VendorRow[]) => {
      const h = vendorHandler([[p.symbol, stockRows], ['KOSPI', benchmarkRows]], index, p.date)
      const feature = buildFeatureVector(h, p.symbol, p.date)
      const context = buildTechnicalContextMap({ handler: h, symbols: [p.symbol], dates, includeFromDate: p.date }).get(p.date)!.get(p.symbol)!
      const atoms = originalInputs18(feature, context)
      const di = m.signalDates.indexOf(p.date)
      atoms[17] = m.breadthEligible[di] ? m.breadthAbove[di] / m.breadthEligible[di] * 100 : null
      return { atoms, signals: buildSignals(feature), feature, eligible: currentProductionEligible(feature, masters.get(p.symbol) as never, stockRows.find(r => r.trade_date === p.date)) }
    }
    const a = point(rows.filter(r => r.trade_date <= p.date), benchmark.filter(r => r.trade_date <= p.date))
    const future = point([...rows.map(r => r.trade_date > p.date ? mutation(r) : r), { ...mutation(rows.at(-1)!), trade_date: '2099-01-04' }], [...benchmark.map(r => r.trade_date > p.date ? mutation(r) : r), { ...mutation(benchmark.at(-1)!), trade_date: '2099-01-04' }])
    const oldCutoff = index.tradingDays[dateIndex - 319]!
    const ancient = point(rows.map(r => r.trade_date < oldCutoff ? mutation(r) : r), benchmark.map(r => r.trade_date < oldCutoff ? mutation(r) : r))
    deepStrictEqual(a, future); deepStrictEqual(a, ancient)
    strictEqual(a.eligible, true)
    deepStrictEqual(a.atoms, p.values.slice(2, 20).map(v => Number.isNaN(v) ? null : v))
    deepStrictEqual(SIGNAL_NAMES.map(k => a.signals[k]), p.values.slice(20, 27))
    strictEqual(a.feature.averageTurnover20, p.values[27])
    witnesses.push({ row: p.row, symbol: p.symbol, date: p.date, source: 'naver-fchart', original18AndSevenScoresExact: true, currentEligibilityExact: true, futureMutatedAndAppendedInvariant: true, pre320MutationInvariant: true })
  }
  for (const [name, hash] of Object.entries(m.sourceHashes)) strictEqual(sha(readFileSync(`${root}/scripts/stock-picks/${name}`)), hash)
  const report = { passed: true, inputManifestPath: manifestPath, inputManifestSha256: sha(readFileSync(manifestPath)), dataSha256: m.output.sha256, moduleSha256: sha(readFileSync(`${base}/naver-ts-observed.ts`)), sampledStockDates: samples.length, sampledSymbols: wanted.size, witnesses, currentEligibilityDifferences: 0, original18Differences: 0, source7Differences: 0, futureMutationDifferences: 0, pre320MutationDifferences: 0, productSourceChanged: false, labelReads: 0, fits: 0, selections: 0, networkCalls: 0 }
  writeFileSync(`${base}/naver-ts-observed/source-parity-causality-audit.json`, JSON.stringify(report, null, 2) + '\n')
  console.log(JSON.stringify({ passed: true, sampledStockDates: samples.length, sampledSymbols: wanted.size, outputSha256: m.output.sha256, differences: 0 }))
}
main().catch(error => { console.error(error); process.exitCode = 1 })
