import {createRequire} from 'node:module'
import {writeFile,readFile} from 'node:fs/promises'
import {createHash} from 'node:crypto'
const root='/Users/isaac/WebstormProjects/stock-ai-newsletter',out='/tmp/composite-score-experimental-20260930'
const require=createRequire(root+'/package.json');require('dotenv').config({path:root+'/.env.local',quiet:true})
const originalFetch=globalThis.fetch;let blockedWrites=0,successfulExternalWrites=0,unadjustedNext=false
const calls:any[]=[],rawFieldNames=new Set<string>(),actionRows:any[]=[]
globalThis.fetch=async(input,init)=>{
 let url=new URL(input instanceof Request?input.url:String(input));const method=(init?.method??(input instanceof Request?input.method:'GET')).toUpperCase()
 const tokenIssue=url.hostname==='openapi.koreainvestment.com'&&url.pathname==='/oauth2/tokenP'&&method==='POST'
 if(!['GET','HEAD'].includes(method)&&!tokenIssue){blockedWrites++;throw new Error('Read-only exploration blocked external mutation')}
 const isPrice=url.hostname==='openapi.koreainvestment.com'&&url.pathname.includes('chartprice')
 if(isPrice&&unadjustedNext){url.searchParams.set('FID_ORG_ADJ_PRC','1');unadjustedNext=false}
 const start=Date.now();const response=await originalFetch(url.toString(),init)
 if(isPrice){const info:any={symbolCode:url.searchParams.get('FID_INPUT_ISCD'),start:url.searchParams.get('FID_INPUT_DATE_1'),end:url.searchParams.get('FID_INPUT_DATE_2'),adjustment:url.searchParams.get('FID_ORG_ADJ_PRC'),status:response.status,elapsedMs:Date.now()-start}
  try{const raw:any=await response.clone().json();info.rt_cd=raw.rt_cd;info.msg_cd=raw.msg_cd;info.rows=Array.isArray(raw.output2)?raw.output2.length:null
   for(const row of raw.output2??[]){Object.keys(row).forEach(k=>rawFieldNames.add(k));if(row.revl_issu_reas||row.prtt_rate&&Number(row.prtt_rate)!==0)actionRows.push({code:info.symbolCode,adjustment:info.adjustment,date:row.stck_bsop_date,prtt_rate:row.prtt_rate,revl_issu_reas:row.revl_issu_reas,mod_yn:row.mod_yn})}
  }catch{info.parseFailure=true}calls.push(info)
 }
 return response
}
const previous=(d:string)=>{const x=new Date(d+'T00:00:00Z');x.setUTCDate(x.getUTCDate()-1);return x.toISOString().slice(0,10).replaceAll('-','')}
async function main(){
 const startedAt=new Date().toISOString();const {fetchDailyRangePriceRows,fetchIndexDailyRangePriceRows}=await import(root+'/app/archive/_utils/api/kis/client.ts')
 const targets=[{symbol:'KOSPI:005930',name:'삼성전자'},{symbol:'KOSDAQ:086520',name:'에코프로'},{symbol:'KOSDAQ:035900',name:'JYP Ent.'},{symbol:'KOSPI',name:'KOSPI index'}]
 const raw:any[]=[];const samples:any[]=[]
 for(const target of targets){let end='20241231';const from='20220101';const byDate=new Map<string,any>();let pages=0,duplicates=0
  while(end>=from){const start=Date.now();const rows=target.symbol==='KOSPI'?await fetchIndexDailyRangePriceRows('0001',from,end):await fetchDailyRangePriceRows(target.symbol,from,end);pages++
   const valid=rows.filter((r:any)=>r.date.replaceAll('-','')>=from&&r.date.replaceAll('-','')<=end).sort((a:any,b:any)=>a.date.localeCompare(b.date))
   if(!valid.length)break
   for(const r of valid){if(byDate.has(r.date))duplicates++;byDate.set(r.date,{symbol:target.symbol,trade_date:r.date,open:r.open,high:r.high,low:r.low,close:r.close,volume:r.volume,source:'kis'})}
   const next=previous(valid[0].date);if(next>=end)throw new Error('Nondecreasing page boundary');end=next
   const left=Math.max(0,550-(Date.now()-start));if(left)await new Promise(r=>setTimeout(r,left))
   if(valid[0].date.replaceAll('-','')<=from)break
   if(pages>20)throw new Error('Small sample page cap exceeded')
  }
  const rows=[...byDate.values()].sort((a,b)=>a.trade_date.localeCompare(b.trade_date));raw.push([target.symbol,rows]);samples.push({...target,rows:rows.length,pages,firstDate:rows[0]?.trade_date,lastDate:rows.at(-1)?.trade_date,duplicates,invalidOhlc:rows.filter(r=>!(r.open>0&&r.high>=r.open&&r.high>=r.close&&r.low<=r.open&&r.low<=r.close&&r.low>0)).length,zeroVolume:rows.filter(r=>r.volume===0).length})
  console.log(JSON.stringify(samples.at(-1)))
 }
 await new Promise(r=>setTimeout(r,550));unadjustedNext=true
 const originalRows=await fetchDailyRangePriceRows('KOSDAQ:086520','20240401','20240430')
 const adjustedRows=raw.find(([s]:any)=>s==='KOSDAQ:086520')[1]
 const adjustments=originalRows.flatMap((r:any)=>{const a=adjustedRows.find((x:any)=>x.trade_date===r.date);return a&&a.close!==r.close?[{date:r.date,adjustedClose:a.close,unadjustedClose:r.close,adjustedVolume:a.volume,unadjustedVolume:r.volume}]:[]})

 await writeFile(out+'/long-history-probe.json',JSON.stringify(report,null,2)+'\n');console.log(JSON.stringify({completed:true,totalPriceCalls:report.totalPriceCalls,totalRows:report.totalRows,blockedWrites,adjustmentDifferences:adjustments.length,elapsedSeconds:(Date.parse(report.finishedAt)-Date.parse(startedAt))/1000}))
}
main().catch(e=>{console.error(e instanceof Error?e.message:'Probe failed');process.exitCode=1})
