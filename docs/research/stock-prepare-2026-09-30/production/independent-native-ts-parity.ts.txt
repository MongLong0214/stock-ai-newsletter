import { readFileSync, writeFileSync } from 'node:fs'
import { createHash } from 'node:crypto'
import { scoreUtilityModel, validateUtilityModel } from '/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/utility-model'

const base = '/tmp/stock-composite-production-20260930'
const raw = JSON.parse(readFileSync(`${base}/independent-model-snapshot.json`, 'utf8'))
const fixture = JSON.parse(readFileSync(`${base}/independent-native-model-cases.json`, 'utf8'))
// The runtime date adapter is currently being wired by its owner; this audit binds
// the actual predictor to the exact captured trained D5 metadata, without edits.
const model = validateUtilityModel({ ...raw, trainedLabelsThrough: raw.metadata.latestLabelMaturity })
let count = 0
for (const test of fixture.cases) {
  const actual = scoreUtilityModel(model, test.inputs)
  if (actual.utility !== test.utility || actual.score !== test.score) {
    throw new Error(`native/TS prediction mismatch case=${count}: ${JSON.stringify({ actual, expected: test })}`)
  }
  count++
}
const sha = (path: string) => createHash('sha256').update(readFileSync(path)).digest('hex')
const result = {
  scope: 'Actual production TypeScript predictor versus independently refitted native sklearn over observed, all-null/zero, single-missing, every finite threshold equality and adjacent IEEE754 cases',
  cases: count, differences: 0, snapshotModelSha256: fixture.modelSnapshotSha256,
  productionPredictorSha256: sha('/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/utility-model.ts'),
  predictorObservedInputModuleSha256: sha('/Users/isaac/WebstormProjects/stock-ai-newsletter/scripts/stock-picks/observed-inputs.ts'),
  dateAdapter: 'trainedLabelsThrough from captured metadata.latestLabelMaturity; final production artifact adapter not yet reviewed',
  productionEdits: 0, newFitsInThisTSComparison: 0,
}
writeFileSync(`${base}/independent-native-ts-parity.json`, JSON.stringify(result, null, 2))
console.log(JSON.stringify(result))
